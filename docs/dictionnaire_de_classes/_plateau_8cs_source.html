126 public bool VerifierCollision(Coup coup)
127 {
128 Piece piece = ObtenirPiece(coup.PositionDepart);
129
130 if (piece == null || piece is Cavalier || piece is Roi)
131 {
132 return false;
133 }
134
135 int differenceLigne = coup.PositionArrivee.Ligne - coup.PositionDepart.Ligne;
136 int differenceColonne = coup.PositionArrivee.Colonne - coup.PositionDepart.Colonne;
137 int pasLigne = Math.Sign(differenceLigne);
138 int pasColonne = Math.Sign(differenceColonne);
139
